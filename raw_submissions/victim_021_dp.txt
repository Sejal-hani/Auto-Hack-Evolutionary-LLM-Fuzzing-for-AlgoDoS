**PROBLEM STATEMENT**

Given a tree with n vertices, you can choose two vertices a and b once and remove all vertices on the path from a to b, including the vertices themselves. If you choose a=b, only one vertex will be removed. Your task is to find the maximum number of connected components that can be formed after removing the path from the tree.

**LIMITS**

- Time limit per test: 2 seconds
- Memory limit per test: 256 megabytes
- Number of test cases: 1 ≤ t ≤ 10^4
- Size of the tree: 2 ≤ n ≤ 2⋅10^5

**CODE**

```cpp
#include <bits/stdc++.h>
using namespace std;

#define pp pair <int, int>
#define fi first
#define se second
#define yes cout << "YES\n"
#define no cout << "NO\n"
#define loop(i, n) for (int i = 1; i <= n; i++)

const int N = 3e5 + 9;

vector <int> adj[N];
int deg[N];
int n;
int dp[N];

pp edges[N];

void dfs (int u, int p){
    dp[u] = deg[u];
    if (p == 0) dp[u]--;
    else dp[u] -= 2;
    if (dp[u] < 0) dp[u] = 0;
    int mx = 0;
    for (int v : adj[u]){
        if (v == p) continue;
        dfs (v, u);
        mx = max (mx, dp[v]);
    }
    if (p) dp[u] += mx;
}

int run (int node){
    for (int i = 1; i <= n; i++) dp[i] = 0;
    dfs (node, 0);
    int temp1 = 0, temp2 = 0;
    for (int v : adj[node]){
        if (dp[v] > temp1){
            temp2 = temp1; temp1 = dp[v];
        } else if (dp[v] > temp2){
            temp2 = dp[v];
        }
    }
    dp[node] += temp1 + temp2;
    return 1 + dp[node];
}

bool cmp (int x, int y){
    return deg[x] > deg[y];
}

void solve (){
    cin >> n;
    for (int i = 1; i < n; i++){
        cin >> edges[i].fi >> edges[i].se;
        adj[edges[i].fi].push_back (edges[i].se);
        adj[edges[i].se].push_back (edges[i].fi);
        deg[edges[i].fi]++;
        deg[edges[i].se]++;
    }
    int node = 0;
    for (int i = 1; i <= n; i++) if (deg[i] > deg[node]) node = i;    
    vector <int> v;
    for (int i = 1; i <= n; i++) v.push_back (i);
    sort (v.begin (), v.end (), cmp);
    int res = 0;
    for (int i = 0; i < v.size () && i < 100; i++){
        res = max (res, run (v[i]));
    }
    cout << res << '\n';
    for (int i = 1; i <= n; i++){
        adj[i].clear ();
        dp[i] = 0;
        deg[i] = 0;
    }
}

signed main (){
    ios_base::sync_with_stdio (false);
    cin.tie (NULL);
    cout.tie (NULL);
    if (fopen ("input.txt", "r")){
        freopen ("input.txt", "r", stdin);
        freopen ("output.txt", "w", stdout);
    }
    int tests = 1; cin >> tests;
    while (tests--) solve ();
}
```
