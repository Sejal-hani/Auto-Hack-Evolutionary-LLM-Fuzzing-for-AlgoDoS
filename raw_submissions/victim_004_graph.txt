**PROBLEM STATEMENT**

You are given an n x m grid, where some cells are filled and some are empty. A city is a maximal set of filled cells such that it is possible to get from any cell in the set to any other cell in the set by moving to adjacent cells, without moving into any cells not in the set. Initially, there are two cities on the grid. You want to change some empty cells into filled cells so that both of the following are satisfied:

- There is one city on the resulting grid.
- The shortest path between any two filled cells, achievable only by moving onto filled cells, is equal to the Manhattan distance between them.

Find a way to add filled cells that satisfies these conditions and minimizes the total number of filled cells.

**LIMITS**

- Time limit per test: 1 second
- Memory limit per test: 256 megabytes
- Number of test cases: 1 ≤ t ≤ 5000
- Grid size: 1 ≤ n, m ≤ 50, nm ≥ 3
- Grid cells: '#' if filled, '.' if empty

**CODE**

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

const int MAX_N = 55;
const int MAX_M = 55;

int n, m;
char grid[MAX_N][MAX_M];
int dx[] = {-1, 0, 1, 0};
int dy[] = {0, -1, 0, 1};

void dfs(int x, int y) {
    grid[x][y] = '#';
    for (int i = 0; i < 4; i++) {
        int nx = x + dx[i];
        int ny = y + dy[i];
        if (nx >= 1 && nx <= n && ny >= 1 && ny <= m && grid[nx][ny] == '.') {
            dfs(nx, ny);
        }
    }
}

void solve() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> grid[i][j];
        }
    }

    int cnt = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (grid[i][j] == '#') {
                dfs(i, j);
                cnt++;
            }
        }
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (grid[i][j] == '.') {
                int x = i, y = j;
                bool ok = true;
                for (int k = 0; k < 4; k++) {
                    int nx = x + dx[k];
                    int ny = y + dy[k];
                    if (nx >= 1 && nx <= n && ny >= 1 && ny <= m && grid[nx][ny] == '#') {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    grid[i][j] = '#';
                    cnt++;
                }
            }
        }
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cout << grid[i][j];
        }
        cout << endl;
    }
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
```
