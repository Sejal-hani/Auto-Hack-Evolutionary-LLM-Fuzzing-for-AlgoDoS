**PROBLEM STATEMENT**

You are given a tree with n nodes, where each node has a value. The tree is rooted at node 1. You are allowed to destroy any edge in the tree, and the cost of destroying an edge is equal to the sum of the values of the two nodes connected by that edge. Your goal is to find the minimum cost to destroy the tree.

**LIMITS**

- The tree has n nodes, where 1 ≤ n ≤ 10^5.
- The values of the nodes are integers between 1 and 10^9.
- The tree is rooted at node 1.

const int N = 2e5 + 5, inf = 1e9;
int n;
vector<int> g[N];
int dp[N];

void dfs(int u, int p) {
    dp[u] = 0;
    for (int v : g[u]) {
        if (v == p) continue;
        dfs(v, u);
        dp[u] = max(dp[u], dp[v]);
    }
    dp[u]++;
}

void solve() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        g[i].clear();
    }
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1, 1);
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        ans = max(ans, dp[i]);
    }
    cout << n - ans << endl;
}

int32_t main() {
    ios_base::sync_with_stdio(false), cin.tie(NULL);
    int tc = 1;
    cin >> tc;
    for (int i = 1; i <= tc; i++) {
        // cout<<"Case "<<i<<": ";
        solve();
    }
