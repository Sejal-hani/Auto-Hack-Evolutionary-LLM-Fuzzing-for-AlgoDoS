**PROBLEM STATEMENT**

Gellyfish hates math problems, but she has to finish her math homework. She is given an array of n positive integers a1, a2, …, an. She needs to do the following two-step operation until all elements of a are equal:

1. Select two indexes i, j satisfying 1≤i,j≤n and i≠j.
2. Replace ai with gcd(ai, aj).

Now, Gellyfish asks you for the minimum number of operations to achieve her goal.

**LIMITS**

- Time limit per test: 2 seconds
- Memory limit per test: 512 megabytes

**CODE**

```cpp
#include <bits/stdc++.h>
#define endl '\n'
#define int long long
#define INF 0x3f3f3f3f3f3f3f3f
using namespace std;

void solve()
{
    int n, t = 0;
    cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; ++i)
    {
        cin >> a[i];
        t = gcd(t, a[i]);
    }

    int cnt = 0;
    unordered_map<int, int> mp;
    for (int i = 1; i <= n; ++i)
    {
        a[i] /= t;
        if (a[i] == 1)
            ++cnt;
        unordered_map<int, int> nmp = mp;
        nmp[a[i]] = 0;
        for (auto [k, v] : mp)
        {
            int nk = gcd(k, a[i]), nv = v + 1;
            if (nmp.count(nk))
                nmp[nk] = min(nmp[nk], nv);
            else
                nmp[nk] = nv;
        }
        mp.swap(nmp);
    }
    if (cnt)
        cout << n - cnt << endl;
    else
        cout << mp[1] + n - 1 << endl;
}

signed main()
{
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t;
    cin >> t;
    while (t--) solve();
}
```
